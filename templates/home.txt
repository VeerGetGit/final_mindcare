// Initial index
let index = 0;

// Setup initial state
updateContent();
updateArrows();

// Function: move forward
function increment() {
  if (index < img_arr.length - 1) {
    index++;
    updateContent();
    updateArrows();
  }
}

// Function: move backward
function decrement() {
  if (index > 0) {
    index--;
    updateContent();
    updateArrows();
  }
}

// Function: update displayed content
function updateContent() {
  scrolle.src = img_arr[index];
  img_cont.textContent = content_arr[index];
  head.textContent = head_arr[index];
}

// Function: control arrow styling
function updateArrows() {
  // Left arrow
  if (index === 0) {
    left.style.color = "rgb(180, 180, 180)";
    left.style.cursor = "not-allowed";
  } else {
    left.style.color = "#00756a";
    left.style.cursor = "pointer";
  }

  // Right arrow
  if (index === img_arr.length - 1) {
    right.style.color = "rgb(180, 180, 180)";
    right.style.cursor = "not-allowed";
  } else {
    right.style.color = "#00756a";
    right.style.cursor = "pointer";
  }
}

// Optional: smooth scroll for Learn More button
function focus_1() {

}
