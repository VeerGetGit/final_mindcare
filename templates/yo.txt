    const streakSnap = await getDoc(streakRef);

// NEW: Check if streak should be reset based on actual completion history
async function checkAndResetStreakIfNeeded(streakData) {
  const today = new Date().toISOString().split('T')[0];
  const currentStreak = streakData.currentStreak || 0;
  
  if (currentStreak === 0) {
    return { shouldNotify: false, newStreak: 0 };
  }

  // Find the last day when all habits were actually completed
  let lastCompletedDate = null;
  let daysMissed = 0;
  
  // Check backwards from yesterday to find the actual last completion
  for (let i = 1; i <= currentStreak + 7; i++) { // Check a reasonable range
    const checkDate = new Date(Date.now() - (i * 24 * 60 * 60 * 1000)).toISOString().split('T')[0];
    
    try {
      const habitsRef = doc(db, "users", currentUser.uid, "habits", checkDate);
      const habitsSnap = await getDoc(habitsRef);
      
      if (habitsSnap.exists()) {
        const dayHabits = habitsSnap.data().habits || {};
        const allCompleted = Object.values(dayHabits).every(habit => habit.completed);
        
        if (allCompleted) {
          lastCompletedDate = checkDate;
          daysMissed = i - 1; // Days between last completion and today
          break;
        }
      }
    } catch (error) {
      console.warn(`Could not check habits for ${checkDate}`);
    }
  }

  // If we found a gap in completions
  if (lastCompletedDate) {
    const lastCompletedDateObj = new Date(lastCompletedDate);
    const todayObj = new Date(today);
    const actualDaysMissed = Math.floor((todayObj - lastCompletedDateObj) / (24 * 60 * 60 * 1000)) - 1;
    
    if (actualDaysMissed > 0) {
      return {
        shouldNotify: true,
        daysMissed: actualDaysMissed,
        lastCompletedDate: lastCompletedDate,
        newStreak: 0 // Reset streak to 0
      };
    }
  }

  return { shouldNotify: false, newStreak: currentStreak };
}

// NEW: Calculate new streak based on actual completion history
async function calculateNewStreak(streakData, today) {
  const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString().split('T')[0];
  let newStreak = 1; // Default to 1 for today
  let wasReset = false;

  if (!streakData) {
    return { newStreak: 1, wasReset: false };
  }

  const currentStreak = streakData.currentStreak || 0;
  
  // Check if yesterday was completed
  try {
    const yesterdayRef = doc(db, "users", currentUser.uid, "habits", yesterday);
    const yesterdaySnap = await getDoc(yesterdayRef);
    
    if (yesterdaySnap.exists()) {
      const yesterdayHabits = yesterdaySnap.data().habits || {};
      const yesterdayCompleted = Object.values(yesterdayHabits).every(habit => habit.completed);
      
      if (yesterdayCompleted) {
        // Yesterday was completed, continue streak
        newStreak = currentStreak + 1;
      } else {
        // Yesterday was not completed, reset streak
        newStreak = 1;
        wasReset = true;
      }
    } else {
      // No data for yesterday means streak is broken
      if (currentStreak > 0) {
        wasReset = true;
      }
      newStreak = 1;
    }
  } catch (error) {
    console.warn('Could not check yesterday completion, starting fresh');
    newStreak = 1;
  }

  return { newStreak, wasReset };
}
